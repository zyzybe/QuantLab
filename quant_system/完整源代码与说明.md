## 一、系统概览

在搜索栏输入股票代码或名称，系统接入 akshare 历史行情，输出净值曲线、四项必需指标、与基准的横向对比及一份基于回测数值的投资研究建议。数据流如下：

```text
浏览器  --POST /api/backtest-->  app.py（参数校验 / 结果缓存 / 异常哨兵）
                                   |
                                   v
                             pipeline.run_pipeline(ctx)
                                   |
   [1] datafeed    akshare 有序源链 -> as_of 钳制 -> 原子缓存 -> market_data
   [2] signals     4 位并行分析员 -> 多空辩论 -> 裁判 -> 风控 -> decision_series
   [3] backtest    t+1 开盘成交 / 手续费 / 整手 / 止损冷却 -> 三条净值 + trades
   [4] metrics     四项必需指标 + 基准横向对比
   [4c] advisory   五维评分 / 结论分档 / 研究动作 / 合规护栏
   [4b] reporting  元信息白名单 + 降级说明 + 免责声明 -> 单一 JSON
                                   |
                                   v
                     app.js 渲染 KPI / 建议卡 / SVG 净值曲线 / 对比表 / 明细
```

**技术栈**：Python 3.12 + Flask + akshare + pandas / numpy + 原生 JavaScript（自绘 SVG，无 CDN）。数据只走本地缓存与 akshare，无数据库、无账号体系。

**说明**

Flask 应用的唯一入口，负责把内部流水线暴露为 HTTP 接口，不承载任何业务逻辑。

```python
QuantLab —— 动态量化回测 Web 应用（Flask 后端）
================================================================
启动：  python app.py

  3 回测引擎   quant/backtest.py   t+1 开盘成交 + 手续费/整手/止损冷却
  4 结果输出   quant/metrics.py + quant/reporting.py  净值曲线 + 四大指标 + 基准横向对比

  结构化输出 / 元信息白名单 / 运行签名 / 防未来函数。
"""
from __future__ import annotations

import threading
import time
import traceback
from datetime import date, timedelta
from pathlib import Path

from flask import Flask, jsonify, request, send_from_directory

from quant.contracts import APP_VERSION, Sentinel
from quant.datafeed import BENCHMARKS, DEFAULT_BENCHMARK, load_universe, search_symbols
from quant.pipeline import build_context, run_pipeline
from quant.reporting import error_report

ROOT = Path(__file__).resolve().parent
STATIC = ROOT / "static"
RESULT_CACHE_TTL = 600
_result_cache: dict[str, tuple[float, dict]] = {}
_cache_lock = threading.Lock()

app = Flask(__name__, static_folder=str(STATIC), static_url_path="/static")


def _json(payload: dict, status_code: int = 200):
    resp = jsonify(payload)
    resp.headers["Cache-Control"] = "no-store"
    return resp, status_code


@app.after_request
def _no_cache(resp):
    resp.headers.setdefault("Cache-Control", "no-store")
    return resp


@app.get("/")
def index():
    return send_from_directory(str(STATIC), "index.html")


@app.get("/api/health")
def health():
    return _json({
        "status": Sentinel.OK,
        "app": "QuantLab",
        "version": APP_VERSION,
        "akshare": "已接入",
        "universe_count": load_universe()["count"],
        "benchmarks": [{"code": b["code"], "name": b["name"]} for b in BENCHMARKS],
    })


@app.get("/api/config")
def config():
    today = date.today()
    return _json({
        "default_symbol": "600519",
        "default_start": (today - timedelta(days=365 * 3)).isoformat(),
        "default_end": today.isoformat(),
        "benchmark": DEFAULT_BENCHMARK,
        "benchmarks": [{"code": b["code"], "name": b["name"]} for b in BENCHMARKS],
        "parameters": {
            "initial_capital": 1000000, "max_position": 0.95, "stop_loss": 0.12,
            "cooldown_days": 5, "fast_window": 20, "slow_window": 60,
            "rsi_window": 14, "momentum_window": 20, "bollinger_window": 20,
            "volume_window": 20, "vote_threshold": 0.25, "rf": 0.02,
        },
    })


@app.get("/api/universe")
def universe():
    q = request.args.get("q", "").strip()
    try:
        return _json({"status": Sentinel.OK, "results": search_symbols(q),
                      "source": load_universe()["source"]})
    except Exception as exc:
        return _json(error_report(Sentinel.DATA_UNAVAILABLE,
                                  f"名录查询失败：{type(exc).__name__}", [str(exc)[:200]]), 200)


@app.route("/api/backtest", methods=["POST", "GET"])
def backtest():
    payload = dict(request.get_json(silent=True) or {}) if request.method == "POST" else dict(request.args)
    ctx, err = build_context(payload)
    if err:
        return _json(error_report("BAD_INPUT", err), 400)

    signature = ctx.signature()
    with _cache_lock:
        hit = _result_cache.get(signature)
        if hit and time.time() - hit[0] < RESULT_CACHE_TTL:
            cached = dict(hit[1])
            cached["from_cache"] = True
            return _json(cached)

    try:
        report = run_pipeline(ctx)
    except Exception as exc:
        # 任一未分类异常都转成哨兵响应，不把堆栈/本地路径泄露给调用方（3.7）
        traceback.print_exc()
        report = error_report(
            Sentinel.DATA_UNAVAILABLE,
            "回测流水线内部异常，未产出任何净值或指标",
            [f"{type(exc).__name__}: {str(exc)[:200]}"],
        )
        report["meta"] = {"run_signature": signature, "as_of": ctx.as_of.isoformat(),
                          "app_version": APP_VERSION}

    if report.get("status") == Sentinel.OK:
        with _cache_lock:
            _result_cache[signature] = (time.time(), report)
    return _json(report, 200 if report.get("status") == Sentinel.OK else 200)


def _warmup_universe() -> None:
    try:
        load_universe()
    except Exception:
        pass


if __name__ == "__main__":
    threading.Thread(target=_warmup_universe, daemon=True).start()
    print(f"QuantLab {APP_VERSION} 启动： http://127.0.0.1:5000")
    app.run(host="127.0.0.1", port=5000, debug=False, threaded=True, use_reloader=False)
```

### 4.3 `quant/contracts.py`

**职责**：状态契约 / 哨兵 / 运行上下文　|　**行数**：205

**说明**

全系统的「类型地基」。它把《TradingAgents 可借鉴设计要点》里的工程约束固化成可检查的类型，而不是散落在注释里。

包含四组内容：`Sentinel` 定义 OK / DATA_UNAVAILABLE / NO_DATA_AVAILABLE / COVERAGE_GAP / REVIEW 等哨兵；`INSTRUCTIONS` 为每个哨兵配一条「调用方接下来该做什么」的指令串；`Decision` / `VendorErrorKind` 把业务动作与数据源错误按「调用方的不同反应数」分类；`RunContext` 是一个显式传参对象，覆盖标的、区间、基准、全部策略参数、费率与 `as_of`，其 `signature()` 方法把影响产出的输入拼成可复现的运行签名；`RunState` 用具名字段声明每个节点写入哪份交付物，节点之间不会互相覆盖 key。

```python
class _Throttle:
    """同类数据源的最小调用间隔，避免东方财富 push2 限流。"""

    def __init__(self, min_interval: float) -> None:
        self.min_interval = min_interval
        self._last = 0.0
        self._lock = threading.Lock()

    def wait(self) -> None:
        with self._lock:
            gap = time.monotonic() - self._last
            if gap < self.min_interval:
                time.sleep(self.min_interval - gap)
            self._last = time.monotonic()

def classify_exception(exc: BaseException) -> VendorErrorKind:
    """按调用方应该怎么反应分类，而不是按成分类。"""
    name = type(exc).__name__
    msg = str(exc)
    if isinstance(exc, (ValueError, TypeError, KeyError)) and "数据" in msg:
        return VendorErrorKind.NO_MARKET_DATA
    if any(k in name for k in ("RateLimit", "Timeout", "Connection", "Proxy", "HTTP")):
        return VendorErrorKind.VENDOR_UNAVAILABLE
    if any(k in msg for k in ("限流", "断开", "超时", "Max retries", "timed out")):
        return VendorErrorKind.VENDOR_UNAVAILABLE
    if isinstance(exc, AttributeError) or "更名" in msg:
        return VendorErrorKind.VENDOR_NOT_CONFIGURED
    return VendorErrorKind.VENDOR_UNAVAILABLE

def _normalize_frame(raw: pd.DataFrame, column_map: dict[str, str], vendor: str) -> tuple[pd.DataFrame, int]:
    """统一为英文列；无效行直接剔除并记录，绝不用插值编造数值。"""
    usable = {k: v for k, v in column_map.items() if k in raw.columns}
    if "date" not in usable.values() or "close" not in usable.values():
        raise ValueError(f"{vendor}: 返回结构缺少必要字段 date/close，实际列={list(raw.columns)}")
    out = raw[list(usable)].rename(columns=usable).copy()
    out["date"] = pd.to_datetime(out["date"], errors="coerce").dt.strftime("%Y-%m-%d")
    for col in ("open", "high", "low", "close", "volume", "amount"):
        if col not in out.columns:
            out[col] = float("nan")
        out[col] = pd.to_numeric(out[col], errors="coerce")
    out = out.dropna(subset=["date"])
    out = out.sort_values("date").drop_duplicates("date", keep="last")
    before = len(out)
    valid = (out["close"] > 0) & (out["open"] > 0) & (out["high"] > 0) & (out["low"] > 0)
    out = out[valid]
    return out[PRICE_COLUMNS].reset_index(drop=True), before - len(out)


def _clip_to_as_of(df: pd.DataFrame, start: date, end: date, as_of: date) -> pd.DataFrame:
    """P0-2：无论上游返回多少，末端一律钳到 as_of 之前，杜绝未来函数。"""
    upper = min(end, as_of).isoformat()
    lower = start.isoformat()
    return df[(df["date"] >= lower) & (df["date"] <= upper)].reset_index(drop=True)

def fetch_price_series(
    kind: str,
    code: str,
    start: date,
    end: date,
    as_of: date,
    adjust: str = "qfq",
) -> DataResult:
    """按显式有序链取数；链穷尽后返回带指令的哨兵，绝不返回空表冒充成功。"""
    if kind not in ("stock", "benchmark"):
        return _failed(Sentinel.DATA_UNAVAILABLE, ["未知数据种类"])

    bench_meta = get_benchmark(code) if kind == "benchmark" else None
    cache_key = f"{kind}_{code}_{adjust}_{start.isoformat()}_{end.isoformat()}_{as_of.isoformat()}"
    cache_path = _cache_file(cache_key)
    cached = _read_cache(cache_path, PRICE_CACHE_TTL)
    if cached is not None and not cached.empty:
        clipped = _clip_to_as_of(cached, start, end, as_of)
        if len(clipped) >= 2:
            meta = _read_cache_meta(cache_path)
            origin = meta.get("vendor", "disk-cache")
            coverage = _coverage(clipped, start, end)
            coverage["dropped_invalid_rows"] = int(meta.get("dropped_invalid_rows", 0))
            cache_notes = [f"命中本地磁盘缓存，原始数据来自 {origin}，本次未产生外部请求"]
            cache_notes.extend(meta.get("notes", []))
            return DataResult(
                status=Sentinel.OK, frame=clipped, vendor=origin,
                coverage=coverage, notes=cache_notes,
            )

    chain = STOCK_VENDORS if kind == "stock" else BENCH_VENDORS
    notes: list[str] = []
    for spec in chain:
        name = spec["name"]
        try:
            _THROTTLES[spec["throttle"]].wait()
            if kind == "stock":
                raw = spec["fetch"](code, start, end, adjust)
            else:
                raw = spec["fetch"](bench_meta, start, end)
            if raw is None or len(raw) == 0:
                notes.append(f"[{name}] 返回空数据 -> NoMarketDataError（与标的有关，停止回落）")
                return _failed(Sentinel.NO_DATA_AVAILABLE, notes)
            normalized, dropped = _normalize_frame(raw, spec["map"], name)
            clipped = _clip_to_as_of(normalized, start, end, as_of)
            if len(clipped) < 2:
                notes.append(f"[{name}] as_of 钳制后不足 2 个交易日 -> NoMarketDataError")
                return _failed(Sentinel.NO_DATA_AVAILABLE, notes)
            coverage = _coverage(clipped, start, end)
            coverage["dropped_invalid_rows"] = dropped
            if dropped:
                notes.append(f"[{name}] 剔除 {dropped} 行无效/缺失行情，未做任何插值补齐")
            if spec["vendor"] == "eastmoney":
                notes.append(f"[{name}] 东财 push2 已节流调用")
            _atomic_write_csv(normalized, cache_path)
            _write_cache_meta(cache_path, {
                "vendor": name, "dropped_invalid_rows": dropped, "notes": notes,
            })
            return DataResult(status=Sentinel.OK, frame=clipped, vendor=name,
                              coverage=coverage, notes=notes)
        except Exception as exc:  # 工具内异常不抛出，避免中断整条流水线
            kind_err = classify_exception(exc)
            short = str(exc).replace("\n", " ")[:160]
            notes.append(f"[{name}] {kind_err.value}: {short}")
            if kind_err == VendorErrorKind.NO_MARKET_DATA:
                break
            continue

    notes.append(INSTRUCTIONS[Sentinel.DATA_UNAVAILABLE])
    return _failed(Sentinel.DATA_UNAVAILABLE, notes)

def _universe_from_exchanges() -> tuple[list[dict], list[str]]:

def load_universe() -> dict:

    path = CACHE_DIR / "universe.json"
    try:
        if path.exists() and time.time() - path.stat().st_mtime < UNIVERSE_CACHE_TTL:
            payload = json.loads(path.read_text(encoding="utf-8"))
            if payload.get("count", 0) > 100:
                return payload
    except Exception:
        pass

    items, notes = _universe_from_exchanges()
    dedup: dict[str, dict] = {}
    for item in items:
        dedup[item["code"]] = item
    merged = list(dedup.values())
    source = "akshare 沪深交易所官方名录"
    if len(merged) < 100:
        merged = [{"code": c, "name": n, "market": market_prefix(c).upper()}

    }

        tmp.write_text(json.dumps(payload, ensure_ascii=False), encoding="utf-8")
        os.replace(tmp, path)
    except OSError:
        pass
    return payload


def search_symbols(query: str, limit: int = 12) -> list[dict]:
    q = _clean_name(query).upper()
    if not q:
        return []
    universe = load_universe()
    pool = universe["items"]
    exact = [x for x in pool if x["code"] == q]
    prefix = [x for x in pool if x["code"].startswith(q) and x not in exact]
    named = [x for x in pool if q in _clean_name(x["name"]).upper() and x not in exact and x not in prefix]
    result = (exact + prefix + named)[:limit]
    if q.isdigit() and len(q) == 6 and not any(x["code"] == q for x in result):
        result.append({"code": q, "name": "名录未收录（将按代码直接查询）",
                       "market": market_prefix(q).upper(), "unlisted": True})
    return result
```

### 4.5 `quant/signals.py`

**职责**：模块 2 策略信号　|　**行数**：358

**说明**

模块 2 策略信号，把「多智能体辩论」落地成确定性公式，也是上一轮问答里那张四分析员表格对应的源码。

四位分析员由 `ANALYST_SPECS` 声明式注册（趋势 1.2 / 动量 1.0 / 波动率 0.8 / 量能 0.8），用 `ThreadPoolExecutor` 并行计算、`wait()` 作 barrier，各自只写自己的 `report_key`。每人每天产出三元组：方向 `+1/-1/0`、强度 `0~1`、固定权重。趋势用 MA20/MA60 状态与金叉死叉加成；动量用 Wilder 平滑 RSI(14) 与 ROC(20)，RSI 超买超卖是**反向**判定；波动率用布林带 z 值做均值回归；量能用 MA5/MA20 量比与当日收益率方向，缩量时强度减半。所有指标只用 rolling / ewm 的当日及历史值。

`_research_manager` 把多空证据分别汇总后按 `score = Σ(权重×方向×强度) ÷ Σ(权重×max(强度,0.05))` 裁决，超过 ±`vote_threshold` 才产出 BUY / SELL；可用分析员少于 2 位时输出 `REVIEW` 哨兵而非 HOLD。`_risk_gate` 再叠加高波动降仓（20 日波动率 > 3.5% 时仓位乘 0.6），并明确「风控只调整仓位、不改写方向」。信号在 t 日收盘生成，由回测引擎在 t+1 执行。

```python
"""
模块 2：策略信号（多分析员委员会）
================================================================
设计要点映射：
  P0-1 角色 = 交付物契约   -> 每位分析员只写自己的 report_key，互不覆盖
  3.1  并行子图 + 收窄输出 -> ThreadPoolExecutor 各自独立计算，输出只含自己的序列
  2.2  并行 + barrier      -> wait() 全部完成后才进入 Research Manager
  4.2  两段辩论            -> Bull / Bear 分别汇总证据，Research Manager 裁决
  P0-4 REVIEW 哨兵         -> 数据不足/报告缺失是 REVIEW，绝不降级成 HOLD
  P0-5 显式缺失标记        -> _absent_mark()，缺席段落以文字出现而非空串
  P0-6 结构化输出          -> InvestmentPlan / DailyDecision 具名字段，渲染在 reporting.py
  5-1  结构化字段判定      -> 用 direction/status 枚举，不靠自由文本前缀
  P0-2 point-in-time      -> 指标仅用 rolling/ewm 的当日及历史值，不引用未来数据

重要：信号在 t 日收盘后生成，回测引擎在 t+1 才执行（见 backtest.py）。
"""
from __future__ import annotations

from concurrent.futures import ThreadPoolExecutor, wait
from typing import Any, Callable

import numpy as np
import pandas as pd

from .contracts import Decision, RunContext, Sentinel

OK = Sentinel.OK
NO_DATA = Sentinel.NO_DATA_AVAILABLE
REVIEW = Sentinel.REVIEW


# ---------------------------------------------------------------- 交付物契约表
# 声明式 registry：新增分析员只要加一行数据，不改编排代码（对应 3.5 能力表思路）
ANALYST_SPECS: list[dict[str, Any]] = [
    {"key": "trend", "label": "趋势分析员", "report_key": "trend_report",
     "weight": 1.2, "fn": None},
    {"key": "momentum", "label": "动量分析员", "report_key": "momentum_report",
     "weight": 1.0, "fn": None},
    {"key": "volatility", "label": "波动率分析员", "report_key": "volatility_report",
     "weight": 0.8, "fn": None},
    {"key": "volume", "label": "量能分析员", "report_key": "volume_report",
     "weight": 0.8, "fn": None},
]


def _absent_mark(label: str) -> str:
    """P0-5：缺席段落必须是说明句，不能是空串让下游脑补。"""
    return f"(No {label} report in this run: it is not available, not an empty finding.)"


def _series_frame(df: pd.DataFrame) -> tuple[pd.Series, pd.Series, pd.Series, pd.Series, pd.Series, int]:
    close = df["close"].astype(float)
    high = df["high"].astype(float)
    low = df["low"].astype(float)
    volume = df["volume"].astype(float)
    open_ = df["open"].astype(float)
    n = len(df)
    return close, high, low, volume, open_, n


def _empty_entry(date_str: str, reason: str) -> dict:
    return {"date": date_str, "status": NO_DATA, "direction": None,
            "strength": 0.0, "evidence": reason}


def _build_report(key: str, label: str, entries: list[dict], note: str) -> dict:
    ok_days = sum(1 for e in entries if e["status"] == OK)
    return {
        "agent_key": key, "label": label, "report_key": f"{key}_report",
        "status": OK if ok_days > 0 else NO_DATA,
        "ok_days": ok_days, "total_days": len(entries),
        "note": note, "values": entries,   # 收窄输出：只有自己的序列
    }


# ---------------------------------------------------------------- 四位分析员
def _trend_analyst(df: pd.DataFrame, ctx: RunContext) -> dict:
    """MA 快/慢线：状态方向 + 金叉死叉事件增强。"""
    close, _, _, _, _, n = _series_frame(df)
    fast = close.rolling(ctx.fast_window, min_periods=ctx.fast_window).mean()
    slow = close.rolling(ctx.slow_window, min_periods=ctx.slow_window).mean()
    pf, ps = fast.shift(1), slow.shift(1)
    golden = (pf <= ps) & (fast > slow)
    death = (pf >= ps) & (fast < slow)
    warm = max(ctx.slow_window, ctx.fast_window)
    entries = []
    for i in range(n):
        d = str(df["date"].iloc[i])
        if i < warm or pd.isna(fast.iloc[i]) or pd.isna(slow.iloc[i]) or slow.iloc[i] == 0:
            entries.append(_empty_entry(d, _absent_mark("trend (warm-up window not filled)")))
            continue
        direction = 1 if fast.iloc[i] > slow.iloc[i] else -1
        strength = float(min(abs(fast.iloc[i] / slow.iloc[i] - 1) / 0.05, 1.0))
        event = "金叉" if golden.iloc[i] else ("死叉" if death.iloc[i] else "均线状态延续")
        if golden.iloc[i] or death.iloc[i]:
            strength = float(min(strength + 0.35, 1.0))
        entries.append({
            "date": d, "status": OK, "direction": direction, "strength": round(strength, 4),
            "evidence": f"MA{ctx.fast_window}={fast.iloc[i]:.2f} / MA{ctx.slow_window}={slow.iloc[i]:.2f}，{event}",
        })
    return _build_report("trend", "趋势分析员", entries,
                         f"双均线状态跟踪，预热 {warm} 个交易日；rolling 窗口只含当日及历史收盘价")


def _momentum_analyst(df: pd.DataFrame, ctx: RunContext) -> dict:
    """RSI 超买超卖 + N 日动量方向。"""
    close, _, _, _, _, n = _series_frame(df)
    delta = close.diff()
    alpha = 1.0 / max(ctx.rsi_window, 2)
    gain = delta.clip(lower=0).ewm(alpha=alpha, adjust=False, min_periods=ctx.rsi_window).mean()
    loss = (-delta.clip(upper=0)).ewm(alpha=alpha, adjust=False, min_periods=ctx.rsi_window).mean()
    rs = gain / loss.replace(0, np.nan)
    rsi = 100 - 100 / (1 + rs)
    rsi = rsi.mask(loss == 0, 100.0)                       # 只涨不跌 -> 100
    rsi = rsi.mask((gain == 0) & (loss > 0), 0.0)          # 只跌不涨 -> 0
    rsi = rsi.mask((gain == 0) & (loss == 0), 50.0)        # 完全横盘 -> 中性 50
    roc = close / close.shift(ctx.momentum_window) - 1
    warm = max(ctx.rsi_window + 1, ctx.momentum_window + 1)
    entries = []
    for i in range(n):
        d = str(df["date"].iloc[i])
        if i < warm or pd.isna(rsi.iloc[i]) or pd.isna(roc.iloc[i]):
            entries.append(_empty_entry(d, _absent_mark("momentum (warm-up window not filled)")))
            continue
        r = float(rsi.iloc[i])
        m = float(roc.iloc[i])
        if r > 70:
            direction, tag = -1, "RSI 超买"
        elif r < 30:
            direction, tag = 1, "RSI 超卖"
        else:
            direction = 1 if m > 0 else (-1 if m < 0 else 0)
            tag = f"{ctx.momentum_window}日动量 {'向上' if m > 0 else ('向下' if m < 0 else '走平')}"
        strength = float(min(abs(r - 50) / 50 + 0.2, 1.0))
        entries.append({
            "date": d, "status": OK, "direction": direction, "strength": round(strength, 4),
            "evidence": f"RSI{ctx.rsi_window}={r:.1f}（{tag}），ROC={m * 100:.2f}%",
        })
    return _build_report("momentum", "动量分析员", entries,
                         "Wilder 平滑 RSI + N 日收益率；超买超卖优先于动量方向")


def _volatility_analyst(df: pd.DataFrame, ctx: RunContext) -> dict:
    """布林带均值回归：偏离中轨越远，反向信号越强。"""
    close, _, _, _, _, n = _series_frame(df)
    w = ctx.bollinger_window
    mid = close.rolling(w, min_periods=w).mean()
    std = close.rolling(w, min_periods=w).std(ddof=0)
    denom = (2 * std).replace(0, np.nan)
    z = (close - mid) / denom
    entries = []
    for i in range(n):
        d = str(df["date"].iloc[i])
        if i < w or pd.isna(z.iloc[i]):
            entries.append(_empty_entry(d, _absent_mark("volatility (warm-up window not filled)")))
            continue
        zv = float(z.iloc[i])
        direction = 1 if zv < -0.4 else (-1 if zv > 0.4 else 0)
        strength = float(min(abs(zv), 1.0))
        entries.append({
            "date": d, "status": OK, "direction": direction, "strength": round(strength, 4),
            "evidence": f"布林带 z={zv:.2f}（中轨 {mid.iloc[i]:.2f}，带宽 2σ）",
        })
    return _build_report("volatility", "波动率分析员", entries,
                         f"布林带({w}, 2σ) 均值回归，|z|>0.4 触发方向")


def _volume_analyst(df: pd.DataFrame, ctx: RunContext) -> dict:
    """量能确认：放量上涨/放量下跌，缩量则降低置信度。"""
    close, _, _, volume, _, n = _series_frame(df)
    vol_s = volume.rolling(5, min_periods=5).mean()
    vol_l = volume.rolling(ctx.volume_window, min_periods=ctx.volume_window).mean()
    ratio = vol_s / vol_l.replace(0, np.nan)
    ret1 = close / close.shift(1) - 1
    warm = max(ctx.volume_window, 6)
    entries = []
    for i in range(n):
        d = str(df["date"].iloc[i])
        if i < warm or pd.isna(ratio.iloc[i]) or pd.isna(ret1.iloc[i]):
            entries.append(_empty_entry(d, _absent_mark("volume (warm-up window not filled)")))
            continue
        rv, rr = float(ratio.iloc[i]), float(ret1.iloc[i])
        direction = 1 if rr > 0 else (-1 if rr < 0 else 0)
        strength = float(min(abs(rv - 1) / 0.5, 1.0))
        if rv < 1.0:  # 缩量：方向保留但置信度打折
            strength *= 0.5
        entries.append({
            "date": d, "status": OK, "direction": direction, "strength": round(strength, 4),
            "evidence": f"量比(5/{ctx.volume_window})={rv:.2f}，当日涨跌 {rr * 100:.2f}%",
        })
    return _build_report("volume", "量能分析员", entries,
                         "短长量均线比确认价格方向，缩量减半置信度")

    "momentum": _momentum_analyst,
    "volatility": _volatility_analyst,
    "volume": _volume_analyst,
}
for _spec in ANALYST_SPECS:
    _spec["fn"] = _ANALYST_FUNCS[_spec["key"]]


# ---------------------------------------------------------------- 编排
def _run_analysts(df: pd.DataFrame, ctx: RunContext) -> tuple[dict[str, dict], list[str]]:
    """并行子图 + barrier：全部分析员交付完成后才进入研究阶段。"""
    notes: list[str] = []
    reports: dict[str, dict] = {}
    with ThreadPoolExecutor(max_workers=len(ANALYST_SPECS)) as pool:
        futures = {pool.submit(spec["fn"], df, ctx): spec for spec in ANALYST_SPECS}
        done, _ = wait(futures)  # barrier
        for fut in done:
            spec = futures[fut]
            key = spec["report_key"]
            try:
                reports[key] = fut.result()  # 每位分析员只写自己的 key
            except Exception as exc:
                # 工具内异常不抛出中断流水线，改为带指令的哨兵报告（3.4(a)）
                notes.append(f"{spec['label']} 计算失败：{type(exc).__name__}: {str(exc)[:120]}；"
                             f"{_absent_mark(spec['key'])}")
                n = len(df)
                reports[key] = _build_report(
                    spec["key"], spec["label"],
                    [_empty_entry(str(df["date"].iloc[i]), _absent_mark(spec["key"]))
                     for i in range(n)],
                    "分析员不可用，本轮全部标记为 REVIEW，不编造信号",
                )
    return reports, notes


def _research_manager(reports: dict[str, dict], ctx: RunContext) -> dict:
    """两段辩论（Bull ⇄ Bear）→ 裁判产出结构化 InvestmentPlan。"""
    first = next(iter(reports.values()))
    n = first["total_days"]
    dates = [e["date"] for e in first["values"]]
    plans = []
    for i in range(n):
        available, absent = [], []
        for spec in ANALYST_SPECS:
            entry = reports[spec["report_key"]]["values"][i]
            if entry["status"] == OK:
                available.append((spec, entry))
            else:
                absent.append(_absent_mark(spec["key"]))
        if len(available) < 2:
            # 哨兵：未判定不是 HOLD，禁止下游当作业务动作
            plans.append({
                "date": dates[i], "status": REVIEW, "direction": Decision.REVIEW.value,
                "score": None, "confidence": 0.0,
                "bull_case": _absent_mark("bull evidence"),
                "bear_case": _absent_mark("bear evidence"),
                "review_reason": f"仅 {len(available)} 位分析员可用（<2），本轮未判定；" + " ".join(absent[:2]),
            })
            continue
        weighted = sum(s["weight"] * e["direction"] * e["strength"] for s, e in available)
        total_weight = sum(s["weight"] * max(e["strength"], 0.05) for s, e in available)
        score = float(weighted / total_weight) if total_weight else 0.0
        bull = [f"{s['label']}:{e['evidence']}" for s, e in available if e["direction"] > 0]
        bear = [f"{s['label']}:{e['evidence']}" for s, e in available if e["direction"] < 0]
        if score > ctx.vote_threshold:
            direction = Decision.BUY.value
        elif score < -ctx.vote_threshold:
            direction = Decision.SELL.value
        else:
            direction = Decision.HOLD.value
        plans.append({
            "date": dates[i], "status": OK, "direction": direction,
            "score": round(score, 4),
            "confidence": round(min(len(available) / len(ANALYST_SPECS), 1.0) * min(abs(score) * 2 + 0.2, 1.0), 4),
            "bull_case": "；".join(bull) if bull else _absent_mark("bull evidence"),
            "bear_case": "；".join(bear) if bear else _absent_mark("bear evidence"),
            "review_reason": "",
        })
    return {"status": OK, "debate_rounds": 1, "analyst_count": len(ANALYST_SPECS),
            "threshold": ctx.vote_threshold, "values": plans}


def _risk_gate(plan: dict, df: pd.DataFrame, ctx: RunContext) -> tuple[list[dict], dict, list[str]]:
    """风控三方（激进 / 保守 / 中立）结构化折中，产出最终交易决策序列。"""
    close = df["close"].astype(float)
    vol20 = (close / close.shift(1) - 1).rolling(20, min_periods=20).std()
    notes: list[str] = []
    decisions: list[dict] = []
    prev_target = 0.0
    reduced_days = 0
    for i, p in enumerate(plan["values"]):
        if p["status"] == REVIEW:
            # REVIEW 不映射为任何业务动作：target_position 为 None，仅留可见标记
            decisions.append({
                "date": p["date"], "decision": Decision.REVIEW.value,
                "target_position": None, "score": None, "confidence": 0.0,
                "status": REVIEW, "bull_case": p["bull_case"], "bear_case": p["bear_case"],
                "review_reason": p["review_reason"], "risk_note": "未判定，不产生新动作",
            })
            continue
        if p["direction"] == Decision.BUY.value:
            target = ctx.max_position                      # 激进派
        elif p["direction"] == Decision.SELL.value:
            target = 0.0
        else:

            risk_note = f"保守派主张降仓（20日波动率 {v:.2%} > 3.5%）；三方折中后仓位 {target:.0%}"
            reduced_days += 1
        prev_target = target
        decisions.append({
            "date": p["date"], "decision": p["direction"],
            "target_position": round(float(target), 4),
            "score": p["score"], "confidence": p["confidence"],

        "high_volatility_reduced_days": reduced_days,

    }

def generate_signals(state: dict, ctx: RunContext) -> dict:
    """模块 2 入口：写入 state 的四份分析员报告 / 投资计划 / 决策序列。"""
    df = state["market_data"]
    reports, notes = _run_analysts(df, ctx)
    for key, report in reports.items():
        state[key] = report                      # 具名交付物，互不覆盖
    plan = _research_manager(reports, ctx)
    state["investment_plan"] = plan
    decisions, risk_note, gate_notes = _risk_gate(plan, df, ctx)
    state["decision_series"] = decisions
    state["risk_note"] = risk_note
    notes.extend(gate_notes)

    review_days = sum(1 for d in decisions if d["decision"] == Decision.REVIEW.value)
    if review_days:
        notes.append(
            f"共 {review_days} 个交易日为 REVIEW（多为指标预热期或分析员不可用），"
            f"这些日期不产生任何交易动作，也不计入业务信号统计。"
        )
    summary = {
        "total_days": len(decisions),
        "buy_days": sum(1 for d in decisions if d["decision"] == Decision.BUY.value),
        "sell_days": sum(1 for d in decisions if d["decision"] == Decision.SELL.value),
        "hold_days": sum(1 for d in decisions if d["decision"] == Decision.HOLD.value),
        "review_days": review_days,
        "analysts": [
            {"key": r["agent_key"], "label": r["label"], "status": r["status"],
             "ok_days": r["ok_days"], "note": r["note"]}
            for r in reports.values()
        ],
    }
    return {"summary": summary, "notes": notes}
```

```python
"""
模块 3：回测引擎
================================================================
设计要点映射：
  P0-2 / 3.4 防未来函数   -> t 日收盘生成信号，t+1 开盘价成交；绝不用当日收盘价成交
  P0-4 REVIEW 哨兵         -> REVIEW 日执行 no_new_action：跳过本次调仓、不计入业务信号、留可见标记
  4.2  结构化字段          -> decision/target_position 用枚举与数值，不解析自由文本
  3.6  评估定位克制        -> 明确这是带手续费/整手/T+1 的简化执行模型，不是真实撮合
  P0-5 显式缺失标记        -> 基准与个股交易日不对齐时只取交集，绝不前向填充编造价格

执行规则：
  * 初始资金、佣金（含最低 5 元）、卖出印花税、100 股整手
  * 每日最多调仓一次（开盘），止损在收盘判定、次日开盘强平，天然满足 T+1
  * 止损后进入冷却期，期间不再开仓
  * 基准为指数收盘价归一化（不含交易成本），买入持有为同起点同仓位规则
"""
from __future__ import annotations

from dataclasses import dataclass, field
from typing import Any

import numpy as np
import pandas as pd

from .contracts import Decision, RunContext, Sentinel


@dataclass
class BacktestResult:
    dates: list[str] = field(default_factory=list)
    strategy_equity: list[float] = field(default_factory=list)
    benchmark_equity: list[float] = field(default_factory=list)
    buy_hold_equity: list[float] = field(default_factory=list)
    benchmark_close: list[float] = field(default_factory=list)
    trades: list[dict] = field(default_factory=list)
    review_no_action_days: int = 0
    force_sell_count: int = 0
    aligned_out_dates: list[str] = field(default_factory=list)
    execution_notes: list[str] = field(default_factory=list)
    final_position: float = 0.0
    status: str = Sentinel.OK


def _fee(amount: float, side: str, ctx: RunContext) -> float:
    commission = max(amount * ctx.commission_rate, ctx.min_commission)
    return commission + (amount * ctx.stamp_tax if side == "SELL" else 0.0)


def _buy_qty(budget: float, price: float, cash: float, ctx: RunContext) -> tuple[float, float, bool]:
    """整手优先；若一手都买不起则启用零股演示模式，并显式返回降级标记。

    A 股真实约束是 100 股整手，但高价股（如 1500 元的茅台）配合小额初始资金会导致
    整段回测 0 交易。这里不静默变成"不买"，而是降级并留可见说明。
    """
    lots = int(budget / (price * 100))
    qty = float(lots * 100) if lots > 0 else budget / price * 0.99
    fractional = lots == 0
    for _ in range(30):
        if qty <= 0:
            break
        amount = qty * price
        fee = _fee(amount, "BUY", ctx)
        if amount + fee <= cash:
            break
        qty *= 0.97
    amount = qty * price
    fee = _fee(amount, "BUY", ctx) if qty > 0 else 0.0
    return qty, fee, fractional


def run_backtest(
    full_market: pd.DataFrame,
    decision_series: list[dict],
    bench_close_map: dict[str, float],
    report_start: str,
    ctx: RunContext,
) -> BacktestResult:
    """full_market 含预热期；信号按日期索引，只在 report_start 之后执行。"""
    result = BacktestResult()
    all_dates = [str(d) for d in full_market["date"]]
    open_ = full_market["open"].astype(float).to_numpy()
    close = full_market["close"].astype(float).to_numpy()
    low = full_market["low"].astype(float).to_numpy()
    decision_map = {d["date"]: d for d in decision_series}

    # 基准与个股交易日只取交集；不对齐的日期显式记录，绝不 ffill 编造
    eligible = [i for i, d in enumerate(all_dates)
                if d >= report_start and d in bench_close_map and bench_close_map[d] > 0]
    excluded = [d for i, d in enumerate(all_dates)
                if d >= report_start and (d not in bench_close_map or bench_close_map.get(d, 0) <= 0)]
    result.aligned_out_dates = excluded
    if excluded:
        result.execution_notes.append(
            f"基准与个股交易日不对齐，剔除 {len(excluded)} 个日期（未做插值/前向填充）："
            f"{', '.join(excluded[:5])}{' ...' if len(excluded) > 5 else ''}"
        )
    if len(eligible) < 2:
        result.status = Sentinel.NO_DATA_AVAILABLE
        result.execution_notes.append("对齐后不足 2 个交易日，无法回测；不编造净值")
        return result

    first = eligible[0]
    cash = float(ctx.initial_capital)
    shares = 0
    avg_cost = 0.0
    prev_target = 0.0
    cooldown_until = -1
    force_sell = False
    base0 = float(bench_close_map[all_dates[first]])

    # 买入持有：同一天开盘、同一仓位规则、同样手续费，保证横向可比
    bh_price = float(open_[first])
    bh_shares, bh_fee, bh_fractional = _buy_qty(
        ctx.initial_capital * ctx.max_position, bh_price, ctx.initial_capital, ctx)
    bh_cash = ctx.initial_capital - bh_shares * bh_price - bh_fee
    fractional_days = 0
    if bh_fractional:
        result.execution_notes.append(
            "该标的一手金额超过可用资金，买入持有与策略均启用零股演示模式"
            "（非 A 股真实整手约束，仅用于保证高价股可回测）")

    for i in eligible:
        d = all_dates[i]
        px_open = float(open_[i])
        px_close = float(close[i])
        equity_before = cash + shares * px_open

        # ---- 开盘：执行 t-1 收盘后生成的信号（防未来函数的关键） ----
        prev_date = all_dates[i - 1] if i > 0 else None
        signal = decision_map.get(prev_date) if prev_date else None
        target = prev_target
        reason = "无前一日信号，保持既有状态"
        if force_sell:
            target, reason = 0.0, "止损强平（触发于上一交易日收盘）"
            force_sell = False
        elif signal is not None and signal.get("status") == Sentinel.REVIEW:
            # REVIEW 不是业务动作：跳过本次调仓，不改变任何状态，单独留痕
            result.review_no_action_days += 1
            reason = "REVIEW：未判定，不产生新动作（未计入业务信号统计）"
        elif signal is not None and signal.get("target_position") is not None:
            target = float(signal["target_position"])
            reason = f"前一日({prev_date})信号 {signal['decision']}，得分 {signal.get('score')}"
        if i < cooldown_until:
            target = 0.0
            reason += "；止损冷却期内不开仓"

        if target is not None and abs(target - prev_target) > 1e-9:
            desired_value = equity_before * target
            current_value = shares * px_open
            if desired_value > current_value + 1:  # 买入
                budget = min(desired_value - current_value, cash)
                qty, fee, fractional = _buy_qty(budget, px_open, cash, ctx)
                if qty > 0:
                    amount = qty * px_open
                    cash -= amount + fee
                    avg_cost = (avg_cost * shares + amount) / (shares + qty)
                    shares += qty
                    if fractional:
                        fractional_days += 1
                    result.trades.append({
                        "date": d, "side": "BUY", "price": round(px_open, 3),
                        "shares": round(qty, 2), "amount": round(amount, 2), "fee": round(fee, 2),
                        "target_position": round(target, 4), "reason": reason,
                    })
            elif current_value > desired_value + 1:  # 卖出
                qty = shares if target <= 0 else min((current_value - desired_value) / px_open, shares)
                qty = min(max(qty, 0), shares)
                if qty > 0:
                    amount = qty * px_open
                    fee = _fee(amount, "SELL", ctx)
                    cash += amount - fee
                    shares -= qty
                    if shares <= 1e-9:
                        shares, avg_cost = 0.0, 0.0
                    result.trades.append({
                        "date": d, "side": "SELL", "price": round(px_open, 3),
                        "shares": round(qty, 2), "amount": round(amount, 2), "fee": round(fee, 2),
                        "target_position": round(target, 4), "reason": reason,
                    })
        prev_target = target if target is not None else prev_target

        # ---- 收盘：结算净值、判定止损 ----
        equity = cash + shares * px_close
        result.dates.append(d)
        result.strategy_equity.append(round(float(equity), 2))
        result.benchmark_equity.append(round(float(ctx.initial_capital * bench_close_map[d] / base0), 2))
        result.benchmark_close.append(round(float(bench_close_map[d]), 4))
        result.buy_hold_equity.append(round(float(bh_cash + bh_shares * px_close), 2))

        if shares > 0 and avg_cost > 0 and (px_close / avg_cost - 1) < -ctx.stop_loss:
            force_sell = True
            cooldown_until = i + ctx.cooldown_days
            result.force_sell_count += 1
            result.execution_notes.append(f"{d} 收盘触发止损（浮亏 {(px_close / avg_cost - 1):.2%}），次日开盘强平")

    result.final_position = round(float(shares * close[eligible[-1]] /
                                        (cash + shares * close[eligible[-1]])), 4) if (cash + shares * close[eligible[-1]]) > 0 else 0.0
    if fractional_days:
        result.execution_notes.append(
            f"共 {fractional_days} 次买入因一手金额超过可用资金而启用零股演示模式")
    result.execution_notes.insert(0, (
        f"执行模型：t 日收盘信号 → t+1 开盘价成交；佣金 {ctx.commission_rate:.4%}（最低 {ctx.min_commission:.0f} 元）、"
        f"卖出印花税 {ctx.stamp_tax:.4%}、100 股整手优先（不足一手时零股兜底并标注）、同日不反向交易。"
    ))
    return result
```

```python
"""
模块 4a：绩效指标
================================================================
必须报告：累计收益率、年化收益率、最大回撤、夏普比率。
设计要点映射：
  P0-4 哨兵而非默认值 -> 样本不足或波动为 0 时返回 None（前端显示"—"），绝不填 0
  3.6  评估定位克制   -> 只陈述历史净值序列的计算结果，不作收益评价或预测
"""
from __future__ import annotations

import math
from typing import Any

import numpy as np

from .contracts import DEFAULT_RF, Sentinel, TRADING_DAYS

CORE_KEYS = ("cumulative_return", "annualized_return", "max_drawdown", "sharpe_ratio")


def compute_metrics(equity: list[float], rf: float = DEFAULT_RF) -> dict[str, Any]:
    eq = np.asarray([float(x) for x in equity], dtype=float)
    out: dict[str, Any] = {
        "status": Sentinel.OK,
        "trading_days": 0,
        "cumulative_return": None,
        "annualized_return": None,
        "max_drawdown": None,
        "sharpe_ratio": None,
        "annualized_volatility": None,
        "calmar_ratio": None,
        "note": "",
    }
    if eq.size < 2 or eq[0] <= 0 or not np.all(np.isfinite(eq)):
        out["status"] = Sentinel.NO_DATA_AVAILABLE
        out["note"] = "净值序列不足或含无效值，四项指标不可用（未以 0 填充）"
        return out

    daily = eq[1:] / eq[:-1] - 1
    n = int(daily.size)
    total = float(eq[-1] / eq[0] - 1)
    years = n / TRADING_DAYS
    annual = float((eq[-1] / eq[0]) ** (1 / years) - 1) if years > 0 and eq[-1] > 0 else None
    peak = np.maximum.accumulate(eq)
    max_dd = float(np.min(eq / peak - 1))
    sd = float(np.std(daily, ddof=1)) if n > 1 else 0.0
    annual_vol = float(sd * math.sqrt(TRADING_DAYS)) if sd > 0 else None
    sharpe = float((np.mean(daily) - rf / TRADING_DAYS) / sd * math.sqrt(TRADING_DAYS)) if sd > 0 else None

    out.update({
        "trading_days": n + 1,
        "cumulative_return": total,
        "annualized_return": annual,
        "max_drawdown": max_dd,
        "sharpe_ratio": sharpe,
        "annualized_volatility": annual_vol,
        "calmar_ratio": float(annual / abs(max_dd)) if annual is not None and max_dd < 0 else None,
    })
    if years < 0.25:
        out["note"] = "回测区间不足一个季度，年化收益率与夏普比率的参考价值有限"
    if sharpe is None:
        out["note"] = (out["note"] + "；日收益波动为 0，夏普比率不可用（未以 0 填充）").strip("；")
    return out


def compare_metrics(strategy: dict, benchmark: dict) -> dict[str, Any]:
    """策略 vs 基准的横向对比：四项必需指标逐项给出差值。"""
    comparison: dict[str, Any] = {}
    for key in CORE_KEYS:
        s, b = strategy.get(key), benchmark.get(key)
        comparison[key] = {
            "strategy": s,
            "benchmark": b,
            "difference": (float(s) - float(b)) if s is not None and b is not None else None,
        }
    comparison["excess_return"] = comparison["cumulative_return"]["difference"]
    comparison["annualized_alpha"] = comparison["annualized_return"]["difference"]
    comparison["note"] = "差值为策略减基准；基准为指数收盘价归一化，不含交易成本"
    return comparison
```

```python
模块 4c：基于回测数据的策略评估与投资研究建议
================================================================
定位（重要）：
  本模块的输出对象是「策略参数组合 + 指定历史区间 + 指定基准」的**历史模拟结果**，
  不是对任何证券的投资建议，不包含买入/卖出/持有意见，不作行情预测或估值判断。
  所有结论仅来自本次回测已产出的数值，由固定阈值规则生成，可逐条复核。

设计要点映射（沿用《TradingAgents 可借鉴设计要点》）：
  P0-4 哨兵而非默认值   -> 指标为 None 时标注"证据不足"并扣分，绝不按 0 参与评分
  P0-5 显式缺失标记     -> 样本/交易/数据不足时直接落到 insufficient 档，不形成倾向性结论
  P0-6 结构化输出       -> verdict / score / highlights / actions / risk_flags 具名字段
  3.6  评估定位克制     -> 只陈述历史计算结果与研究流程建议
  3.7  报告即观测       -> generated_from 带运行签名，评分规则随报告一并输出
  0.   合规护栏         -> guardrail 对所有分档恒定存在，任何分数都不导向实盘或买卖
"""
from __future__ import annotations

import math
from typing import Any

from .backtest import BacktestResult
from .contracts import RunContext, Sentinel
from .reporting import DISCLAIMER

CORE = ("cumulative_return", "annualized_return", "max_drawdown", "sharpe_ratio")

GUARDRAIL = (

# ---------------------------------------------------------------- 小工具
def _f(v: Any) -> float | None:
    """None / NaN / 非数值一律返回 None，绝不静默变成 0（P0-4）。"""
    if v is None:
        return None
    try:
        x = float(v)
    except (TypeError, ValueError):
        return None
    return x if math.isfinite(x) else None


def _pct(v: Any, digits: int = 2) -> str:
    x = _f(v)
    return "—" if x is None else f"{x * 100:+.{digits}f}%"


def _num(v: Any, digits: int = 2) -> str:
    x = _f(v)
    return "—" if x is None else f"{x:+.{digits}f}"


def _ge(v: float, pairs: list[tuple[float, float]], default: float) -> float:
    """越大越好： thresholds 降序。"""
    for threshold, score in pairs:
        if v >= threshold:
            return score
    return default


def _le(v: float, pairs: list[tuple[float, float]], default: float) -> float:
    """越小越好： thresholds 升序。"""
    for threshold, score in pairs:
        if v <= threshold:
            return score
    return default


def _level(score: float, maximum: float, unknown: bool = False) -> str:
    if unknown:
        return "unknown"
    ratio = score / maximum if maximum else 0
    return "good" if ratio >= 0.7 else ("mid" if ratio >= 0.4 else "weak")


def _dim(key: str, score: float, evidence: str, advice: str,
         unknown: bool = False) -> dict[str, Any]:
    label, maximum = DIMENSION_WEIGHT[key]
    return {
        "key": key, "label": label,
        "score": round(float(score), 1), "max": maximum,
        "level": _level(score, maximum, unknown),
        "evidence": evidence, "advice": advice,
    }


# ---------------------------------------------------------------- 五个维度
def _score_return(m: dict, ctx: RunContext, missing: list[str]) -> dict[str, Any]:
    total, annual = _f(m.get("cumulative_return")), _f(m.get("annualized_return"))
    if total is None or annual is None:
        missing.append("绝对收益维度：年化或累计收益率不可用")
        return _dim("return", 6.0,
                    "年化/累计收益率由指标层返回 null（未按 0 填充），本维度按证据不足处理。",
                    "先确认净值序列长度与有效性，再重新运行回测。", unknown=True)
    score = _ge(annual, [(0.15, 25), (0.08, 20), (ctx.rf, 14), (0.0, 9), (-0.10, 5)], 1)
    evidence = (f"累计收益率 {_pct(total)}，年化收益率 {_pct(annual)}；"
                f"对比无风险利率 {_pct(ctx.rf)}。")
    advice = ("绝对收益达到规则中的较高档位，但仍需确认它并非由单一区间或少数几笔交易贡献。"
              if score >= 14 else
              "绝对收益未稳定超过无风险利率，优先复核信号逻辑与仓位规则，而非调高阈值追求曲线。")
    return _dim("return", score, evidence, advice)


def _score_risk(m: dict, missing: list[str]) -> dict[str, Any]:
    sharpe, dd = _f(m.get("sharpe_ratio")), _f(m.get("max_drawdown"))
    vol, calmar = _f(m.get("annualized_volatility")), _f(m.get("calmar_ratio"))
    score = 0.0
    unknown = False
    if sharpe is None:
        score += 3.0
        unknown = True
        missing.append("风险调整维度：夏普比率为 null（日收益波动为 0 或样本不足）")
    else:
        score += _ge(sharpe, [(1.5, 15), (1.0, 12), (0.5, 8), (0.0, 5), (-0.5, 2)], 0.5)
    if dd is None:
        score += 2.0
        unknown = True
        missing.append("风险调整维度：最大回撤不可用")
    else:
        score += _le(abs(dd), [(0.10, 10), (0.20, 7), (0.30, 4), (0.45, 2)], 0.5)
    evidence = (f"夏普比率 {_num(sharpe)}，最大回撤 {_pct(dd)}，"
                f"年化波动率 {_pct(vol)}，Calmar {_num(calmar)}。")
    if unknown:
        advice = "存在不可用指标，本维度按证据不足扣分；不要用 0 代替缺失值后重算。"
    elif score >= 17:
        advice = "风险调整后表现相对均衡，可继续检验回撤是否集中在个别极端区间。"
    else:
        advice = "回撤或波动消耗了收益，建议做止损线、仓位上限与冷却期的参数敏感性网格。"
    return _dim("risk", score, evidence, advice, unknown)


def _score_relative(comparison: dict, missing: list[str]) -> dict[str, Any]:
    excess = _f(comparison.get("excess_return"))
    alpha = _f(comparison.get("annualized_alpha"))
    sharpe_diff = _f((comparison.get("sharpe_ratio") or {}).get("difference"))
    dd_diff = _f((comparison.get("max_drawdown") or {}).get("difference"))
    unknown = False
    if excess is None or alpha is None:
        unknown = True
        missing.append("相对基准维度：超额收益或年化 alpha 不可用")
        score = 6.0
    else:
        score = _ge(excess, [(0.10, 13), (0.03, 10), (0.0, 7), (-0.05, 4), (-0.15, 2)], 0.5)
        score += _ge(alpha, [(0.05, 6), (0.02, 4.5), (0.0, 3), (-0.05, 1.5)], 0.5)
    score += (_ge(sharpe_diff, [(0.5, 3), (0.0, 2)], 0.5) if sharpe_diff is not None else 1.5)
    # dd_diff = 策略回撤 - 基准回撤；为正表示策略回撤更浅
    score += (_ge(dd_diff, [(0.0, 3), (-0.05, 1.5)], 0.5) if dd_diff is not None else 1.5)
    evidence = (f"累计超额 {_pct(excess)}，年化 alpha {_pct(alpha)}，"
                f"夏普差 {_num(sharpe_diff)}，回撤差 {_pct(dd_diff)}（差值＝策略−基准）。")
    advice = ("相对基准占优，仍需换用不同基准指数复核，避免结论依赖单一基准选择。"
              if (excess is not None and excess > 0) else
              "相对基准落后时，应先区分「策略逻辑失效」与「本区间风格不匹配」；"
              "无论何种原因，都不构成对标的本身的评价。")
    return _dim("relative", score, evidence, advice, unknown)


def _score_robustness(m: dict, bt: BacktestResult,
                      missing: list[str]) -> dict[str, Any]:
    days = _f(m.get("trading_days")) or 0
    n_trades = len(bt.trades)
    score = _ge(float(days), [(500, 6), (250, 4.5), (120, 3), (60, 1.5)], 0.5)
    if 10 <= n_trades <= 120:
        score += 4
    elif 3 <= n_trades < 10 or 120 < n_trades <= 250:
        score += 2.5
    else:
        score += 1
    review_ratio = bt.review_no_action_days / max(days, 1)
    score += _le(review_ratio, [(0.02, 3), (0.10, 2)], 1)
    aligned = len(bt.aligned_out_dates or [])
    score += 2 if aligned == 0 else (1 if aligned <= 5 else 0.5)
    evidence = (f"有效净值 {int(days)} 个交易日，{n_trades} 笔成交，"
                f"REVIEW 不动作 {bt.review_no_action_days} 天（占比 {review_ratio * 100:.1f}%），"
                f"止损 {bt.force_sell_count} 次，基准未对齐被剔除 {aligned} 天，"
                f"期末仓位 {bt.final_position * 100:.0f}%。")
    if days < 120 or n_trades < 3:
        missing.append("证据充分性：样本交易日不足 120 天或成交少于 3 笔")
        advice = "样本或成交次数太少，统计意义上不足以评估；应延长区间后再看结论是否稳定。"
    else:
        advice = "把区间切成互不重叠的子段分别复跑，检查结论在不同子区间是否一致。"
    return _dim("robustness", score, evidence, advice)


def _score_data(dq: dict, notes: list[str]) -> dict[str, Any]:
    stock, bench = dq.get("stock", {}), dq.get("benchmark", {})
    both_ok = (stock.get("status") == Sentinel.OK and bench.get("status") == Sentinel.OK)
    score = 4.0 if both_ok else 1.0
    cov_s, cov_b = stock.get("coverage", {}), bench.get("coverage", {})
    has_cov = bool(cov_s.get("rows")) and bool(cov_b.get("rows")) and \
              bool(cov_s.get("first_date")) and bool(cov_b.get("first_date"))
    score += 3.0 if has_cov else 1.0
    score += 3.0 if len(notes) <= 3 else (2.0 if len(notes) <= 8 else 1.0)
    dropped = int(cov_s.get("dropped_invalid_rows", 0)) + int(cov_b.get("dropped_invalid_rows", 0))
    evidence = (f"个股源 {stock.get('vendor', 'N/A')}（{stock.get('status', 'N/A')}），"
                f"基准源 {bench.get('vendor', 'N/A')}（{bench.get('status', 'N/A')}）；"
                f"降级/执行留痕 {len(notes)} 条，剔除无效行情 {dropped} 行，"
                f"数据覆盖个股 {cov_s.get('first_date', '—')}~{cov_s.get('last_date', '—')}。")
    advice = ("数据链路完整，仍需留意「基准为指数归一化、不含交易成本」带来的口径差异。"
              if both_ok else "数据源状态异常，请先修复数据链路再解读任何绩效数值。")
    return _dim("data", score, evidence, advice)

def _highlights(m: dict, b: dict, bh: dict, comparison: dict,

    })
    out.append({
        "tone": "neg" if (_f(m.get("max_drawdown")) or 0) < -0.20 else "neu",
        "text": (f"策略最大回撤 {_pct(m.get('max_drawdown'))}（基准 {_pct(b.get('max_drawdown'))}），"
                 f"夏普 {_num(m.get('sharpe_ratio'))}（基准 {_num(b.get('sharpe_ratio'))}）；"
                 f"回撤与波动需与收益一起看，单项数值不构成优劣判断。"),
    })
    out.append({
        "tone": "neu",
        "text": (f"同一标的在同一区间的买入持有累计 {_pct(bh.get('cumulative_return'))}，"

def _actions(m: dict, comparison: dict, bt: BacktestResult,
             days: float, notes: list[str]) -> list[dict[str, str]]:
    acts: list[dict[str, str]] = []

    def add(priority: str, title: str, detail: str) -> None:

            "单一区间跑输既不能证明策略失效，也不能推导标的强弱。")

def _risk_flags(m: dict, bt: BacktestResult, insufficient: bool) -> list[str]:

        "策略跑赢或跑输基准，反映的是规则与区间的匹配程度，不构成对该标的的评价或预测。",

    ]
    if insufficient:
        flags.insert(0, "本次样本/指标不完整，规则引擎已落到「证据不足」档，不形成倾向性结论。")
    if bt.review_no_action_days:
        flags.append(f"区间内有 {bt.review_no_action_days} 个 REVIEW 不动作日，按 no_new_action 处理，不等同于 HOLD。")
    return flags

def build_advisory(
    ctx: RunContext,
    metrics: dict[str, Any],
    comparison: dict[str, Any],
    bt: BacktestResult,
    signal_summary: dict[str, Any],
    state: dict[str, Any],
) -> dict[str, Any]:
    """纯函数：只读取本次回测已产出的结构化字段，不访问网络、不引入外部观点。"""
    m, b, bh = metrics["strategy"], metrics["benchmark"], metrics["buy_hold"]
    dq = state.get("data_quality", {})
    notes = list(state.get("degradation_notes", [])) + list(bt.execution_notes)
    missing: list[str] = []

    dims = [
        _score_return(m, ctx, missing),
        _score_risk(m, missing),
        _score_relative(comparison, missing),
        _score_robustness(m, bt, missing),
        _score_data(dq, notes),
    ]
    total = round(sum(d["score"] for d in dims), 1)

    days = float(_f(m.get("trading_days")) or 0)
    core_missing = any(_f(m.get(k)) is None for k in CORE)
    data_bad = any(dq.get(k, {}).get("status") != Sentinel.OK for k in ("stock", "benchmark"))
    blockers: list[str] = []
    if days < 60:
        blockers.append(f"有效净值仅 {int(days)} 个交易日（门槛 60）")
    if len(bt.trades) < 2:
        blockers.append(f"仅 {len(bt.trades)} 笔成交（门槛 2 笔）")
    if core_missing:
        blockers.append("四项必需指标中存在 null（未按 0 填充）")
    if data_bad:
        blockers.append("个股或基准数据源状态非 OK")
    insufficient = bool(blockers)

    if insufficient:
        level = "insufficient"
    elif total < 42:
        level = "caution"
    elif total < 65:
        level = "neutral"
    else:
        level = "constructive"

    verdict = dict(VERDICTS[level])
    verdict["level"] = level

    highlights = _highlights(m, b, bh, comparison, bt, ctx)
    if insufficient:
        highlights.insert(0, {
            "tone": "neg",
            "text": "未达到评分门槛：以下数值仅作事实陈列，规则引擎不形成综合评分与倾向性结论。",
        })

    return {
        "available": True,
        "title": "回测解读与投资研究建议",
        "scope": (f"评估对象：本策略参数组合在 {ctx.start_date.isoformat()} ~ {ctx.end_date.isoformat()} "
                  f"对 {ctx.name}（{ctx.symbol}）以 {ctx.benchmark_name} 为基准的历史模拟结果。"
                  "评估对象是策略与规则，不是该证券，也不包含任何操作意见。"),
        "verdict": verdict,

                if insufficient else
                "综合评分为五维求和，仅用于归纳本次回测证据，不代表未来表现。"),
            "dimensions": dims,
            "methodology": ("五维固定阈值评分：绝对收益 25 / 风险调整 25 / 相对基准 25 / "
```

### 4.10 `quant/pipeline.py`

**职责**：编排层 / 参数校验　|　**行数**：163

**说明**

编排层，是 API 与命令行共用的唯一执行入口，保证「报告 parity」。

`build_context` 承担全部不可信参数的校验与钳制：股票代码必须是 6 位数字、日期必须为 ISO 格式、区间不超过 12 年、基准必须在白名单内、所有数值参数用 `num()` 夹到允许范围、`fast_window < slow_window`；最关键的 `as_of = min(end, today)` 由服务端注入，前端无法拿到未来数据。实体名称由 `resolve_name` 从名录确定性解析，不信任调用方传来的名字。

`run_pipeline` 按 数据接入 → 策略信号 → 回测引擎 → 指标计算 → 建议生成 → 报告装配 顺序执行，每阶段只写自己的具名交付物；核心数据失败立即返回指令式哨兵，不产出半截数值；`state` 显式传递而非全局配置，使多请求并发互不串参数。

```python
"""
编排层：四模块流水线
================================================================
入口层(API) 与 CLI 共用同一条 run_pipeline，保证"报告 parity"。
设计要点映射：
  6.1 最小可复制骨架 -> RunState 具名状态 / as_of 注入 / 显式降级 / 运行签名
  5-9 显式 RunContext -> 不用全局 set_config，多请求并发互不串配置
  3.7 运行签名       -> signature() 覆盖全部影响产出物的输入
  P0-3 失败语义       -> 核心数据失败返回指令式哨兵，绝不返回半截数值
"""
from __future__ import annotations

from datetime import date, timedelta
from typing import Any

from .advisory import build_advisory
from .backtest import run_backtest
from .contracts import INSTRUCTIONS, RunContext, Sentinel
from .datafeed import DEFAULT_BENCHMARK, fetch_price_series, get_benchmark, load_universe
from .metrics import compare_metrics, compute_metrics
from .reporting import build_report, error_report
from .signals import generate_signals

WARMUP_DAYS = 420          # 自然日，覆盖 60 日均线等最长预热窗口
MIN_START = date(1995, 1, 1)
MAX_SPAN_DAYS = 365 * 12


def resolve_name(symbol: str) -> str:
    """P0-2：实体身份由服务端确定性解析，不随调用方传来的名称漂移。"""
    for item in load_universe()["items"]:
        if item["code"] == symbol:
            return item["name"]
    return "名录未收录标的"


def build_context(payload: dict[str, Any]) -> tuple[RunContext | None, str | None]:
    """校验 + 钳制：所有边界参数在服务端生成，绝不信任前端。"""
    symbol = str(payload.get("symbol", "")).strip()
    if not (symbol.isdigit() and len(symbol) == 6):
        return None, "股票代码必须是 6 位数字（如 600519、000001、300750）"
    try:
        start = date.fromisoformat(str(payload.get("start_date", "")))
        end = date.fromisoformat(str(payload.get("end_date", "")))
    except (ValueError, TypeError):
        return None, "日期格式应为 YYYY-MM-DD"

    today = date.today()
    as_of = min(end, today)              # 服务端注入，前端无法拿到未来数据
    if start < MIN_START:
        return None, f"起始日期不得早于 {MIN_START.isoformat()}"
    if start >= as_of:
        return None, "起始日期必须早于结束日期，且结束日期不得晚于今天"
    if (as_of - start).days > MAX_SPAN_DAYS:
        return None, "单次回测区间不得超过 12 年"

    bench_code = str(payload.get("benchmark", DEFAULT_BENCHMARK))
    bench_meta = get_benchmark(bench_code)
    if bench_meta["code"] != bench_code:
        return None, "不支持的基准指数"

    def num(key: str, default: float, lo: float, hi: float) -> float:
        try:
            v = float(payload.get(key, default))
        except (TypeError, ValueError):
            return default
        return min(max(v, lo), hi)

    ctx = RunContext(
        symbol=symbol,
        name=resolve_name(symbol),
        start_date=start,
        end_date=as_of,
        as_of=as_of,
        benchmark=bench_meta["code"],
        benchmark_name=bench_meta["name"],
        initial_capital=num("initial_capital", 1_000_000.0, 10_000.0, 100_000_000.0),
        commission_rate=num("commission_rate", 0.00025, 0.0, 0.005),
        stamp_tax=num("stamp_tax", 0.0005, 0.0, 0.005),
        max_position=num("max_position", 0.95, 0.1, 1.0),
        stop_loss=num("stop_loss", 0.12, 0.02, 0.5),
        cooldown_days=int(num("cooldown_days", 5, 0, 30)),
        fast_window=int(num("fast_window", 20, 3, 120)),
        slow_window=int(num("slow_window", 60, 10, 250)),
        rsi_window=int(num("rsi_window", 14, 3, 60)),
        momentum_window=int(num("momentum_window", 20, 3, 120)),
        bollinger_window=int(num("bollinger_window", 20, 5, 120)),
        volume_window=int(num("volume_window", 20, 5, 120)),
        vote_threshold=num("vote_threshold", 0.25, 0.0, 1.0),
        rf=num("rf", 0.02, 0.0, 0.1),
    )
    if ctx.fast_window >= ctx.slow_window:
        return None, "快线窗口必须小于慢线窗口"
    return ctx, None


def run_pipeline(ctx: RunContext) -> dict[str, Any]:
    """四模块顺序执行，每阶段只写自己的具名交付物。"""
    state: dict[str, Any] = {
        "run_context": ctx,
        "run_signature": ctx.signature(),
        "as_of": ctx.as_of.isoformat(),
        "date_window": {"start": ctx.start_date.isoformat(), "end": ctx.end_date.isoformat()},
        "degradation_notes": [],
    }

    # ---- 模块 1：数据接入（含预热期，加热数据也受 as_of 钳制） ----
    fetch_start = ctx.start_date - timedelta(days=WARMUP_DAYS)
    stock = fetch_price_series("stock", ctx.symbol, fetch_start, ctx.end_date,
                               ctx.as_of, ctx.adjust)
    bench = fetch_price_series("benchmark", ctx.benchmark, fetch_start, ctx.end_date,
                               ctx.as_of)
    state["data_quality"] = {"stock": stock.as_quality(), "benchmark": bench.as_quality()}
    if not stock.ok:
        return error_report(stock.status,
                            f"个股 {ctx.symbol} 行情不可用：{stock.status}",
                            stock.notes + bench.notes)
    if not bench.ok:
        return error_report(bench.status,
                            f"基准 {ctx.benchmark_name} 行情不可用：{bench.status}",
                            stock.notes + bench.notes)
    state["market_data"] = stock.frame
    state["benchmark_data"] = bench.frame
    state["degradation_notes"].extend(stock.notes)
    state["degradation_notes"].extend(bench.notes)
    bench_close_map = {
        str(row["date"]): float(row["close"])
        for _, row in bench.frame.iterrows()
        if float(row["close"]) > 0
    }

    # ---- 模块 2：策略信号（并行分析员 -> 多空辩论 -> 裁判 -> 风控） ----
    signal_info = generate_signals(state, ctx)
    state["degradation_notes"].extend(signal_info["notes"])
    signal_summary = signal_info["summary"]
    signal_summary["investment_plan"] = {
        "analyst_count": state["investment_plan"]["analyst_count"],
        "debate_rounds": state["investment_plan"]["debate_rounds"],
        "vote_threshold": state["investment_plan"]["threshold"],
    }
    signal_summary["risk_note"] = state["risk_note"]

    # ---- 模块 3：回测引擎 ----
    bt = run_backtest(stock.frame, state["decision_series"], bench_close_map,
                      ctx.start_date.isoformat(), ctx)
    if bt.status != Sentinel.OK:
        return error_report(bt.status, "回测无法完成：交易日对齐后样本不足",
                            bt.execution_notes + state["degradation_notes"])

    # ---- 模块 4：结果输出 ----
    metrics = {
        "strategy": compute_metrics(bt.strategy_equity, ctx.rf),
        "benchmark": compute_metrics(bt.benchmark_equity, ctx.rf),
        "buy_hold": compute_metrics(bt.buy_hold_equity, ctx.rf),
    }
    comparison = compare_metrics(metrics["strategy"], metrics["benchmark"])
    state["equity_curve"] = bt.strategy_equity
    state["metrics"] = metrics
    # 模块 4c：只读取已产出的结构化结果，规则化生成"策略评估 + 研究建议"
    advisory = build_advisory(ctx, metrics, comparison, bt, signal_summary, state)
    state["advisory"] = advisory
    return build_report(ctx, state, bt, signal_summary, metrics, comparison, advisory)
```

```javascript
function renderChart(r) {
  const svg = $('chart');
  const wrap = $('chartWrap');
  const W = Math.max(620, Math.round(wrap.clientWidth || 1000));
  const H = 380;
  const pad = { l: 66, r: 18, t: 18, b: 34 };
  svg.setAttribute('viewBox', `0 0 ${W} ${H}`);
  svg.setAttribute('width', '100%');
  svg.setAttribute('height', String(H));

  const dates = r.curves.dates || [];
  const n = dates.length;
  if (n < 2) { svg.innerHTML = ''; return; }
  const series = [
    { key: 'strategy', data: r.curves.strategy, width: 2.4 },
    { key: 'benchmark', data: r.curves.benchmark, width: 1.8 },
    { key: 'buy_hold', data: r.curves.buy_hold, width: 1.6 },
  ];
  const visible = series.filter((s) => S.visible[s.key]);
  let lo = Infinity, hi = -Infinity;
  visible.forEach((s) => s.data.forEach((v) => {
    if (Number.isFinite(v)) { lo = Math.min(lo, v); hi = Math.max(hi, v); }
  }));
  if (!Number.isFinite(lo)) return;
  const margin = (hi - lo) * 0.08 || Math.max(hi * 0.01, 1);
  lo -= margin; hi += margin;
  const x = (i) => pad.l + (i / (n - 1)) * (W - pad.l - pad.r);
  const y = (v) => pad.t + (1 - (v - lo) / (hi - lo)) * (H - pad.t - pad.b);

  let html = '';
  for (let g = 0; g <= 5; g += 1) {
    const v = lo + (hi - lo) * g / 5;
    const yy = y(v);
    html += `<line x1="${pad.l}" y1="${yy.toFixed(1)}" x2="${W - pad.r}" y2="${yy.toFixed(1)}" stroke="#eef1f6" stroke-width="1"/>`;
    html += `<text x="${pad.l - 10}" y="${(yy + 4).toFixed(1)}" text-anchor="end" font-size="11" fill="#98a2b3">${fmtMoney(v)}</text>`;
  }
  const initial = Number(r.meta.parameters.initial_capital);
  if (initial > lo && initial < hi) {
    const y0 = y(initial);
    html += `<line x1="${pad.l}" y1="${y0.toFixed(1)}" x2="${W - pad.r}" y2="${y0.toFixed(1)}" stroke="#d0d5dd" stroke-width="1" stroke-dasharray="5 4"/>`;
    html += `<text x="${W - pad.r - 4}" y="${(y0 - 6).toFixed(1)}" text-anchor="end" font-size="10.5" fill="#98a2b3">初始资金</text>`;
  }
  for (let k = 0; k < 6; k += 1) {
    const i = Math.round(k * (n - 1) / 5);
    html += `<text x="${x(i).toFixed(1)}" y="${H - 11}" text-anchor="middle" font-size="11" fill="#98a2b3">${esc(dates[i].slice(2))}</text>`;
  }
  visible.forEach((s) => {
    const pts = s.data.map((v, i) => `${x(i).toFixed(2)},${y(v).toFixed(2)}`).join(' ');
    html += `<polyline points="${pts}" fill="none" stroke="${COLORS[s.key]}" stroke-width="${s.width}" stroke-linejoin="round" stroke-linecap="round"/>`;
  });
  html += `<line id="hoverLine" x1="0" y1="${pad.t}" x2="0" y2="${H - pad.b}" stroke="#98a2b3" stroke-width="1" stroke-dasharray="3 3" style="display:none"/>`;
  visible.forEach((s) => {
    html += `<circle id="dot-${s.key}" r="4" fill="${COLORS[s.key]}" stroke="#fff" stroke-width="2" style="display:none"/>`;
  });
  html += `<rect id="hoverCatch" x="${pad.l}" y="${pad.t}" width="${W - pad.l - pad.r}" height="${H - pad.t - pad.b}" fill="transparent"/>`;
  svg.innerHTML = html;

  const hoverLine = $('hoverLine');
  const tooltip = $('tooltip');
  const catcher = $('hoverCatch');
  catcher.addEventListener('mousemove', (e) => {
    const rect = svg.getBoundingClientRect();
    const px = (e.clientX - rect.left) / rect.width * W;
    let i = Math.round((px - pad.l) / (W - pad.l - pad.r) * (n - 1));
    i = Math.max(0, Math.min(n - 1, i));
    const xx = x(i);
    hoverLine.setAttribute('x1', xx.toFixed(1));
    hoverLine.setAttribute('x2', xx.toFixed(1));
    hoverLine.style.display = '';
    visible.forEach((s) => {
      const dot = $(`dot-${s.key}`);
      dot.setAttribute('cx', xx.toFixed(1));
      dot.setAttribute('cy', y(s.data[i]).toFixed(1));
      dot.style.display = '';
    });
    const wrapRect = wrap.getBoundingClientRect();
    tooltip.innerHTML = `<div class="tt-date">${esc(dates[i])}</div>` + visible.map((s) => {
      const ret = s.data[i] / initial - 1;
      return `<div class="tt-row"><span>${esc(SERIES_LABEL[s.key])}</span><b class="${cls(ret)}">${Number(s.data[i]).toFixed(0)} (${fmt(ret, true)})</b></div>`;
    }).join('');
    tooltip.style.left = `${Math.max(8, Math.min(e.clientX - wrapRect.left + 14, wrapRect.width - 196))}px`;
    tooltip.style.top = '18px';
    tooltip.classList.remove('hidden');
  });
  catcher.addEventListener('mouseleave', () => {
    hoverLine.style.display = 'none';
    visible.forEach((s) => { $(`dot-${s.key}`).style.display = 'none'; });
    tooltip.classList.add('hidden');
  });
  S.chart = { W, H, pad, x, y, visible, dates, initial };
}
```
